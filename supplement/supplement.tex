This supplement accompanies the paper \emph{The Cost of Unknown Friction in Minimum-Time Vehicle
Transit}. Section~\ref{supp:evidence} gives the values behind every numerical statement of the
paper, and Section~\ref{supp:transient} gives the proofs of the results of
Section~\mainref{sec:transient}{15} of the paper. Theorems, sections, and equations without the prefix S
refer to the paper.

\section{Numerical evidence}\label{supp:evidence}

In this section we give the values behind every numerical statement of the paper. Each table is
generated by the code that accompanies the paper from the same computations that the text describes,
so that every reported check can be inspected and reproduced.

\subsection{Measured data}

Table~\ref{tab:app-epa} summarises the inversion of the EPA road-load coefficients. Tables
\ref{tab:app-d3mass}--\ref{tab:app-battery} give the D3 results of Section~\mainref{sec:data}{3}, and
Table~\ref{tab:app-060} the $0$--$60$\,mph comparison. Table~\ref{tab:fmvss} lists all $114$
determinations of friction from the FMVSS~135 reports.

\section{Proofs for the transient deficit}\label{supp:transient}

In this section we prove Theorems~\mainref{thm:transientbound}{15.1} and~\mainref{thm:transientvar}{15.2} and
Proposition~\mainref{prop:transientfail}{15.3} of the main text. We first write the penalty as an integral,
and then prove two elementary facts on which all three proofs rest.

Along the lagged run we write $v(t)$ for the speed and $A(t)=\amax(v(t))$ for the envelope at that
speed. The states $y_1,\dots,y_k$ of the cascade start from rest and satisfy
\[
y_j'=\frac{y_{j-1}-y_j}{\tau_j},\qquad y_j(0)=0,\qquad j=1,\dots,k,
\]
with input $y_0=A$. The delivered acceleration is $d=\min(y_k,A)$, so that $v'=d$. Each stage is a
weighted average of its input,
\[
y_j(t)=\frac1{\tau_j}\int_0^te^{-(t-s)/\tau_j}\,y_{j-1}(s)\,ds,
\]
so every $y_j$ is non-negative because $A$ is positive. Hence $d\ge0$, and the speed is
non-decreasing.

\begin{lemma}[The penalty as an integral]\label{lem:penalty}
Let the lagged run reach the target speed $V$ at the time $T_{\mathrm{real}}$. Then
\begin{equation}\label{eq:penalty}
T_{\mathrm{real}}-T_{\mathrm{phys}}=\int_0^{T_{\mathrm{real}}}\Big(1-\frac{d(t)}{A(t)}\Big)\,dt .
\end{equation}
\end{lemma}

\begin{proof}
The time along the envelope is $T_{\mathrm{phys}}=\int_0^Vdw/\amax(w)$. The speed $v$ is
non-decreasing, with $v(0)=0$, $v(T_{\mathrm{real}})=V$, and $v'=d$, so the substitution $w=v(t)$ gives
\[
T_{\mathrm{phys}}=\int_0^{T_{\mathrm{real}}}\frac{d(t)}{\amax(v(t))}\,dt=\int_0^{T_{\mathrm{real}}}\frac{d(t)}{A(t)}\,dt .
\]
Subtracting this from $T_{\mathrm{real}}=\int_0^{T_{\mathrm{real}}}dt$ gives~\eqref{eq:penalty}.
Since $d\le A$, the integrand is non-negative, so the penalty is non-negative.
\end{proof}

The integrand of~\eqref{eq:penalty} is the shortfall of the delivered acceleration relative to the
envelope. The proofs below bound it with two elementary facts. The first is a comparison principle
for a single lag, and the second describes the response of the cascade to a unit step.

\begin{lemma}[Comparison for one lag]\label{lem:gronwall}
Let $\tau>0$, and let $e$ be absolutely continuous on $[t_0,t_1]$ with $e'\ge-e/\tau$ almost
everywhere. Then $e(t)\ge e(t_0)\,e^{-(t-t_0)/\tau}$ for every $t\in[t_0,t_1]$. In particular,
$e(t_0)\ge0$ implies $e\ge0$ on $[t_0,t_1]$.
\end{lemma}

\begin{proof}
The function $e^{t/\tau}e(t)$ has derivative $e^{t/\tau}\big(e'+e/\tau\big)\ge0$, so it is
non-decreasing. Hence $e^{t/\tau}e(t)\ge e^{t_0/\tau}e(t_0)$, which is the claim.
\end{proof}

\begin{lemma}[Step response of the cascade]\label{lem:step}
Let $\zeta_0\equiv1$, and let $\zeta_1,\dots,\zeta_k$ satisfy
$\zeta_j'=(\zeta_{j-1}-\zeta_j)/\tau_j$ with $\zeta_j(0)=0$. Then each $\zeta_j$ is non-decreasing,
with $0\le\zeta_j\le1$ and $\zeta_j(t)\to1$ as $t\to\infty$, and
\[
\int_0^\infty\big(1-\zeta_k(t)\big)\,dt=\sum_{j=1}^k\tau_j .
\]
\end{lemma}

\begin{proof}
We argue by induction on $j$, the case $j=0$ being trivial. Suppose that $\zeta_{j-1}$ is
non-decreasing, takes values in $[0,1]$, and tends to $1$. As above,
\[
\zeta_j(t)=\frac1{\tau_j}\int_0^te^{-(t-s)/\tau_j}\,\zeta_{j-1}(s)\,ds .
\]
This is non-negative, and since $\zeta_{j-1}(s)\le\zeta_{j-1}(t)$ for $s\le t$, it is at most
$\big(1-e^{-t/\tau_j}\big)\zeta_{j-1}(t)\le\zeta_{j-1}(t)\le1$. Therefore
$\zeta_j'=(\zeta_{j-1}-\zeta_j)/\tau_j\ge0$, so $\zeta_j$ is non-decreasing. It tends to $1$ because
it is a stable first-order response to an input that tends to $1$.

For the integral, note that $\zeta_{j-1}-\zeta_j=\tau_j\zeta_j'\ge0$, so
\[
\int_0^\infty\big(\zeta_{j-1}-\zeta_j\big)\,dt=\tau_j\lim_{t\to\infty}\big(\zeta_j(t)-\zeta_j(0)\big)=\tau_j .
\]
The differences telescope, $1-\zeta_k=\sum_{j=1}^k(\zeta_{j-1}-\zeta_j)$, and summing over $j$ gives
the result.
\end{proof}

\begin{proof}[Proof of Theorem~\mainref{thm:transientbound}{15.1}]
We compare the cascade along the run with the step response of Lemma~\ref{lem:step}. Let
$z_j=y_j/A$ for $j=0,\dots,k$, so that $z_0\equiv1$. The function $A$ is absolutely continuous,
because $\amax$ is locally Lipschitz and $v$ is absolutely continuous. Differentiating $z_j$ gives
\begin{align*}
z_j'&=\frac{y_j'}{A}-\frac{y_jA'}{A^2}\\
&=\frac{z_{j-1}-z_j}{\tau_j}-z_j\,\frac{A'}{A}.
\end{align*}
The speed is non-decreasing and $\amax$ is non-increasing, so $A$ is non-increasing and $A'\le0$
almost everywhere. Since $z_j\ge0$, the last term is non-negative, and
\begin{equation}\label{eq:zineq}
z_j'\ge\frac{z_{j-1}-z_j}{\tau_j}.
\end{equation}

We now show that $z_j\ge\zeta_j$ by induction on $j$. Both sides equal $1$ for $j=0$. Suppose that
$z_{j-1}\ge\zeta_{j-1}$. By~\eqref{eq:zineq}, $z_j'\ge(\zeta_{j-1}-z_j)/\tau_j$, while
$\zeta_j'=(\zeta_{j-1}-\zeta_j)/\tau_j$. The difference $e=z_j-\zeta_j$ therefore satisfies
$e'\ge-e/\tau_j$ with $e(0)=0$, and Lemma~\ref{lem:gronwall} gives $e\ge0$.

In particular $z_k\ge\zeta_k$, and $\zeta_k\le1$ by Lemma~\ref{lem:step}. The integrand
of~\eqref{eq:penalty} therefore satisfies
\[
1-\frac dA=1-\min(z_k,1)\le1-\zeta_k .
\]
By Lemmas~\ref{lem:penalty} and~\ref{lem:step},
\[
T_{\mathrm{real}}-T_{\mathrm{phys}}\le\int_0^{T_{\mathrm{real}}}\big(1-\zeta_k\big)\,dt
\le\int_0^\infty\big(1-\zeta_k\big)\,dt=\sum_{j=1}^k\tau_j .
\]

For a constant envelope $A'=0$, so~\eqref{eq:zineq} holds with equality and $z_k=\zeta_k\le1$. The
penalty is then exactly $\int_0^{T_{\mathrm{real}}}(1-\zeta_k)\,dt$, which tends to $\sum_j\tau_j$ as
$T_{\mathrm{real}}$ grows with the target speed.
\end{proof}

\begin{proof}[Proof of Theorem~\mainref{thm:transientvar}{15.2}]
With one lag we write $y=y_1$, and we define the \emph{relative debt} $r=1-y/A$. It starts at
$r(0)=1$, and $r\le1$ because $y\ge0$. Since $d/A=\min(y/A,1)$, the integrand of~\eqref{eq:penalty}
is
\[
1-\frac dA=\max(r,0).
\]

\emph{Step 1: a differential inequality for the debt.} Differentiating $r$ gives
\begin{align*}
r'&=-\frac{y'}{A}+\frac{y\,A'}{A^2}\\
&=-\frac{A-y}{\tau A}+(1-r)\frac{A'}{A}\\
&=-\frac r\tau+(1-r)\frac{A'}{A}.
\end{align*}
Let $\gamma=\max(A'/A,0)$. Wherever $0\le r\le1$, the factor $1-r$ lies in $[0,1]$, so
$(1-r)A'/A\le\gamma$, and
\begin{equation}\label{eq:debt}
r'\le\gamma-\frac r\tau .
\end{equation}

\emph{Step 2: comparison with a majorant.} Let $\rho$ solve $\rho'=\gamma-\rho/\tau$ with
$\rho(0)=1$, that is,
\[
\rho(t)=e^{-t/\tau}+\int_0^te^{-(t-s)/\tau}\,\gamma(s)\,ds .
\]
Both terms are non-negative, so $\rho\ge0$. We claim that $\max(r,0)\le\rho$. Since $\rho\ge0$, it
suffices to show that $r\le\rho$ wherever $r>0$. Suppose instead that $r(t_1)>\rho(t_1)$ for some
$t_1$, and let $t_0$ be the last time before $t_1$ at which $r\le\rho$. Such a time exists because
$r(0)=\rho(0)$, and $r(t_0)=\rho(t_0)$ by continuity. On $(t_0,t_1]$ we have $r>\rho\ge0$, so
\eqref{eq:debt} holds there, and $e=\rho-r$ satisfies $e'\ge-e/\tau$ with $e(t_0)=0$.
Lemma~\ref{lem:gronwall} gives $e(t_1)\ge0$, which contradicts $r(t_1)>\rho(t_1)$.

\emph{Step 3: integration.} Extend $\gamma$ by zero after $T_{\mathrm{real}}$. By
Lemma~\ref{lem:penalty} and Step~2,
\[
T_{\mathrm{real}}-T_{\mathrm{phys}}\le\int_0^{T_{\mathrm{real}}}\rho\,dt\le\int_0^\infty\rho\,dt
=\tau+\tau\int_0^{T_{\mathrm{real}}}\gamma\,dt,
\]
where the last step exchanges the order of integration and uses $\int_0^\infty e^{-t/\tau}dt=\tau$.

\emph{Step 4: the variation.} By the chain rule, $(\ln A)'=(\ln\amax)'(v)\,v'$, and $v'\ge0$, so
$\gamma=\max\big((\ln\amax)'(v),0\big)\,v'$. The substitution $w=v(t)$ then gives
$\int_0^{T_{\mathrm{real}}}\gamma\,dt=\operatorname{Var}^+$, which completes the proof.
\end{proof}

\begin{proof}[Proof of Proposition~\mainref{prop:transientfail}{15.3}]
The relative debt $r=1-y/A$ of the proof of Theorem~\mainref{thm:transientvar}{15.2} is again the integrand
of~\eqref{eq:penalty} wherever $r\ge0$. On each level $A$ is constant, so $r'=-r/\tau$, and $r$ decays
as $e^{-t/\tau}$. A level on which $r$ starts at $r_i\ge0$ and which lasts the time $h_i$ therefore
contributes $\tau r_i(1-e^{-h_i/\tau})$ to the penalty, and this tends to $\tau r_i$ as $h_i$ grows.

The first level starts at $r_0=1$ and contributes $\tau$ in the limit. At the step from $a_{i-1}$ to
$a_i$, the state $y$ is continuous while $A$ jumps, so the debt jumps to
\[
r_i=1-\frac{y}{a_i}=1-\big(1-r(t_i^-)\big)\frac{a_{i-1}}{a_i},
\]
where $r(t_i^-)$ is the debt just before the step. Since $a_{i-1}<a_i$, this gives
$r_i\ge1-a_{i-1}/a_i>0$ whenever $r(t_i^-)\ge0$, so by induction the debt is positive throughout the
run. As the time on level $i-1$ grows, $r(t_i^-)\to0$, and so $r_i\to1-a_{i-1}/a_i$. Summing the contributions of all levels gives the limit. For
$a_i=a_0q^i$, every step contributes $1-1/q$.
\end{proof}

\begin{proof}[Proof of Remark~\mainref{rem:relax}{15.4}]
With $\tau=\ell_r/v$ the lag runs in distance rather than in time. Writing $x$ for the distance
travelled and $a>0$ for the envelope near rest, the lag state satisfies $dy/dx=(a-y)/\ell_r$, so
$y\approx ax/\ell_r$ for small $x$. Since $v\,dv/dx=y$, this gives $v\approx x\sqrt{a/\ell_r}$ near
rest. The time $\int dx/v$ to cover any distance from rest therefore diverges logarithmically at
$x=0$. The floor speed $v_0$ removes this divergence, and the penalty grows without bound as
$v_0\to0$.
\end{proof}
